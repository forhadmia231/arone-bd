// Admin-managed shipment information. One shipment per order in Phase 2.
// No external courier service is contacted and COD collection is manually attested.
enum CourierShipmentStatus {
  DRAFT
  BOOKED
  IN_TRANSIT
  OUT_FOR_DELIVERY
  DELIVERED
  FAILED
  RETURNED
  CANCELLED
}

model CourierShipment {
  id            String                @id @default(cuid())
  orderId       String                @unique
  order         Order                 @relation(fields: [orderId], references: [id], onDelete: Restrict)
  courierName   String
  consignmentId String                @default("")
  trackingUrl   String                @default("")
  status        CourierShipmentStatus @default(DRAFT)
  codExpected   Int
  codCollected  Int                   @default(0)
  version       Int                   @default(0)
  events        CourierShipmentEvent[]
  createdAt     DateTime              @default(now())
  updatedAt     DateTime              @updatedAt

  @@index([status, updatedAt])
}

model CourierShipmentEvent {
  id              String                 @id @default(cuid())
  shipmentId      String
  shipment        CourierShipment        @relation(fields: [shipmentId], references: [id], onDelete: Cascade)
  adminId         String
  admin           User                   @relation(fields: [adminId], references: [id], onDelete: Restrict)
  fromStatus      CourierShipmentStatus?
  toStatus        CourierShipmentStatus
  beforeCollected Int
  afterCollected  Int
  fromCourierName String                 @default("")
  toCourierName   String                 @default("")
  fromConsignment String                 @default("")
  toConsignment   String                 @default("")
  note            String
  createdAt       DateTime               @default(now())

  @@index([shipmentId, createdAt])
  @@index([createdAt])
}
